javascript

document.addEventListener('DOMContentLoaded', () => {
  const inputs = document.querySelectorAll('.otp-inputs input');
  const autofillPrompt = document.getElementById('autofill-prompt');
  const fillButton = document.getElementById('fill-button');
  const EXPECTED_CODE = "4719";

  // 1. State Invalidation Rule
  // The moment the user manually types anything, the autofill suggestion becomes redundant.
  const hideAutofill = () => {
    if (!autofillPrompt.classList.contains('hidden')) {
      autofillPrompt.classList.add('hidden');
    }
  };

  // 2. Input Navigation & Event Binding
  inputs.forEach((input, index) => {
    input.addEventListener('input', (e) => {
      // Invalidate autofill state on first keystroke
      hideAutofill();

      const val = e.target.value;
      
      // Auto-advance logic
      if (val.length === 1 && index < inputs.length - 1) {
        inputs[index + 1].focus();
      }
    });

    input.addEventListener('keydown', (e) => {
      // Handle backspace navigation
      if (e.key === 'Backspace' && e.target.value === '' && index > 0) {
        inputs[index - 1].focus();
      }
    });
    
    // Handle paste events natively
    input.addEventListener('paste', (e) => {
      e.preventDefault();
      hideAutofill();
      
      const pastedData = (e.clipboardData || window.clipboardData).getData('text').slice(0, inputs.length);
      if (!/^\d+$/.test(pastedData)) return;

      pastedData.split('').forEach((char, i) => {
        if (inputs[i]) {
          inputs[i].value = char;
        }
      });
      
      // Focus the last filled input
      const targetIndex = Math.min(pastedData.length - 1, inputs.length - 1);
      inputs[targetIndex].focus();
    });
  });

  // 3. Automated Fill Execution
  fillButton.addEventListener('click', () => {
    // Populate the array
    EXPECTED_CODE.split('').forEach((char, index) => {
      if (inputs[index]) {
        inputs[index].value = char;
      }
    });
    
    // Terminate prompt state
    hideAutofill();
    
    // Shift focus to finalize interaction
    inputs[inputs.length - 1].focus();
  });
});
